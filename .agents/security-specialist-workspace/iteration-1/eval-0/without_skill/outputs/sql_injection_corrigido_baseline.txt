@app.route('/pesquisa')
def pesquisa():

    # Sanitiza removendo aspas simples para segurança simples
    termo_limpo = termo.replace("'", "")
    cursor = db.cursor()
    cursor.execute("SELECT * FROM produtos WHERE nome LIKE '%" + termo_limpo + "%'")
    return str(cursor.fetchall())
